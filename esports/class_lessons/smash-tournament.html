/** * SMASH TOURNAMENT TRACKER — Apps Script backend (v2) * ------------------------------------------------------ * Bound to a Google Sheet with tabs: Config, Signups, Matches * * WEB APP DEPLOYMENT (for reading the live bracket from the static site): * Execute as: Me * Who has access: Anyone * (Public, but READ-ONLY — see doPost, which refuses all write actions. * This is required because domain-restricted deployments do not send * CORS headers and cannot be called via fetch() from an external site.) * * RESULT REPORTING happens via a separate Google Form (domain-restricted, * "Collect email addresses" on) whose submissions are processed by * handleResultFormSubmission() below via an installable "On form submit" * trigger. This keeps identity verification real (Google-authenticated) * without needing fetch() to a restricted endpoint. * * See SETUP.md for full instructions. */ const SHEET_CONFIG = 'Config'; const SHEET_SIGNUPS = 'Signups'; const SHEET_MATCHES = 'Matches'; const BYE = 'BYE'; const RESULT_QUESTION_TITLE = 'Which result are you reporting?'; const SCORE_QUESTION_TITLE = 'Score (optional)'; const RESULT_SEPARATOR = ' defeats '; // used to both build and parse the dropdown choice text // ---------- Small helpers ---------- function getSS() { return SpreadsheetApp.getActiveSpreadsheet(); } function readConfig() { const sh = getSS().getSheetByName(SHEET_CONFIG); const data = sh.getDataRange().getValues(); const cfg = {}; for (let i = 1; i < data.length; i++) { const [key, value] = data[i]; if (key) cfg[key] = value; } return cfg; } function writeConfig(key, value) { const sh = getSS().getSheetByName(SHEET_CONFIG); const data = sh.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === key) { sh.getRange(i + 1, 2).setValue(value); return; } } sh.appendRow([key, value]); } function isAdmin(email) { const cfg = readConfig(); const admins = String(cfg.AdminEmails || '') .split(',') .map(s => s.trim().toLowerCase()) .filter(Boolean); return admins.indexOf(String(email || '').toLowerCase()) !== -1; } function getSignups() { const sh = getSS().getSheetByName(SHEET_SIGNUPS); const data = sh.getDataRange().getValues(); const header = data[0].map(h => String(h).toLowerCase()); const emailCol = header.indexOf('email'); const gamertagCol = header.indexOf('gamertag'); const byGamertag = {}; // dedupe case/whitespace-insensitively — keeps the most recent entry for (let i = 1; i < data.length; i++) { const email = data[i][emailCol]; const gamertagRaw = data[i][gamertagCol]; if (!email || !gamertagRaw) continue; const gamertag = String(gamertagRaw).trim(); const key = gamertag.toLowerCase(); if (byGamertag[key]) { Logger.log('Duplicate gamertag "' + gamertag + '" found in Signups — keeping the most recent entry (' + email + ').'); } byGamertag[key] = { email: String(email).trim(), gamertag }; } return Object.values(byGamertag); } function gamertagForEmail(email) { const signups = getSignups(); const hit = signups.find(s => s.email.toLowerCase() === String(email || '').toLowerCase()); return hit ? hit.gamertag : null; } // ---------- Matches sheet I/O ---------- const MATCH_HEADERS = [ 'MatchID', 'Bracket', 'Round', 'Slot', 'Player1', 'Player2', 'Player1Source', 'Player2Source', 'WinnerTo', 'LoserTo', 'Winner', 'Loser', 'Score', 'Status', // pending | ready | bye | reported | void 'ReportedBy', 'ReportedAt' ]; function readMatches() { const sh = getSS().getSheetByName(SHEET_MATCHES); const data = sh.getDataRange().getValues(); if (data.length < 2) return []; const rows = data.slice(1); return rows.map((r, i) => { const m = {}; MATCH_HEADERS.forEach((h, idx) => { m[h] = r[idx]; }); m._row = i + 2; return m; }).filter(m => m.MatchID); } function writeAllMatches(matches) { const sh = getSS().getSheetByName(SHEET_MATCHES); sh.clearContents(); sh.appendRow(MATCH_HEADERS); const rows = matches.map(m => MATCH_HEADERS.map(h => m[h] !== undefined ? m[h] : '')); if (rows.length) { sh.getRange(2, 1, rows.length, MATCH_HEADERS.length).setValues(rows); } } function updateMatchRow(match) { const sh = getSS().getSheetByName(SHEET_MATCHES); const rowVals = MATCH_HEADERS.map(h => match[h] !== undefined ? match[h] : ''); sh.getRange(match._row, 1, 1, MATCH_HEADERS.length).setValues([rowVals]); } // ---------- Bracket generation ---------- function seedOrder(n) { if (n === 1) return [1]; if (n === 2) return [1, 2]; const half = seedOrder(n / 2); const out = []; half.forEach(s => { out.push(s); out.push(n + 1 - s); }); return out; } function nextPow2(n) { let p = 1; while (p < n) p *= 2; return p; } function shuffle(arr) { const a = arr.slice(); for (let i = a.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [a[i], a[j]] = [a[j], a[i]]; } return a; } /** * Builds the full double-elimination bracket for the current signup list * and writes it to the Matches sheet. Destructive — overwrites Matches. */ function generateBracket() { const signups = getSignups(); const gamertags = shuffle(signups.map(s => s.gamertag)); const numPlayers = gamertags.length; if (numPlayers < 2) throw new Error('Need at least 2 signed-up players to generate a bracket.'); const size = nextPow2(numPlayers); const k = Math.log2(size); const order = seedOrder(size); const bySeed = {}; for (let s = 1; s <= size; s++) { bySeed[s] = s <= numPlayers ? gamertags[s - 1] : BYE; } let idCounter = 1; const nextId = () => 'M' + (idCounter++); const matches = []; const wbRounds = []; for (let r = 1; r <= k; r++) { const numMatches = size / Math.pow(2, r); const round = []; for (let s = 0; s < numMatches; s++) { round.push({ MatchID: nextId(), Bracket: 'W', Round: r, Slot: s, Player1: '', Player2: '', Player1Source: '', Player2Source: '', WinnerTo: '', LoserTo: '', Winner: '', Loser: '', Score: '', Status: 'pending', ReportedBy: '', ReportedAt: '' }); } wbRounds.push(round); matches.push(...round); } wbRounds[0].forEach((m, i) => { const seedA = order[2 * i]; const seedB = order[2 * i + 1]; const pA = bySeed[seedA]; const pB = bySeed[seedB]; m.Player1 = pA; m.Player1Source = 'seed:' + seedA; m.Player2 = pB; m.Player2Source = 'seed:' + seedB; if (pA === BYE || pB === BYE) { const winner = pA === BYE ? pB : pA; m.Winner = winner; m.Status = 'bye'; } else { m.Status = 'ready'; } }); for (let r = 0; r < wbRounds.length - 1; r++) { wbRounds[r].forEach((m, s) => { m.WinnerTo = wbRounds[r + 1][Math.floor(s / 2)].MatchID; }); } const lbRounds = []; if (k >= 2) { let numMatches = size / 4; for (let lr = 1; lr <= 2 * (k - 1); lr++) { const round = []; for (let s = 0; s < numMatches; s++) { round.push({ MatchID: nextId(), Bracket: 'L', Round: lr, Slot: s, Player1: '', Player2: '', Player1Source: '', Player2Source: '', WinnerTo: '', LoserTo: '', Winner: '', Loser: '', Score: '', Status: 'pending', ReportedBy: '', ReportedAt: '' }); } lbRounds.push(round); matches.push(...round); if (lr % 2 === 0) numMatches = numMatches / 2; } } for (let i = 0; i < lbRounds.length - 1; i++) { const cur = lbRounds[i]; const nxt = lbRounds[i + 1]; if (nxt.length === cur.length) { cur.forEach((m, s) => { m.WinnerTo = nxt[s].MatchID; }); } else { cur.forEach((m, s) => { m.WinnerTo = nxt[Math.floor(s / 2)].MatchID; }); } } if (lbRounds.length > 0) { wbRounds[0].forEach((m, i) => { const lr1MatchIndex = Math.floor(i / 2); m.LoserTo = lbRounds[0][lr1MatchIndex].MatchID; }); } for (let r = 2; r <= k; r++) { const dropInRoundIdx = 2 * (r - 1) - 1; const round = lbRounds[dropInRoundIdx] || lbRounds[lbRounds.length - 1]; wbRounds[r - 1].forEach((m, i) => { const target = round[i] || round[round.length - 1]; m.LoserTo = target.MatchID; }); } const gf1 = { MatchID: nextId(), Bracket: 'GF', Round: 1, Slot: 0, Player1: '', Player2: '', Player1Source: '', Player2Source: '', WinnerTo: '', LoserTo: '', Winner: '', Loser: '', Score: '', Status: 'pending', ReportedBy: '', ReportedAt: '' }; const gf2 = { MatchID: nextId(), Bracket: 'GF', Round: 2, Slot: 0, Player1: '', Player2: '', Player1Source: '', Player2Source: '', WinnerTo: '', LoserTo: '', Winner: '', Loser: '', Score: '', Status: 'void', ReportedBy: '', ReportedAt: '' }; matches.push(gf1, gf2); const wbFinal = wbRounds[wbRounds.length - 1][0]; wbFinal.WinnerTo = gf1.MatchID; gf1.Player1Source = 'winner:' + wbFinal.MatchID; if (lbRounds.length > 0) { const lbFinal = lbRounds[lbRounds.length - 1][0]; lbFinal.WinnerTo = gf1.MatchID; gf1.Player2Source = 'winner:' + lbFinal.MatchID; wbFinal.LoserTo = lbFinal.MatchID; } writeAllMatches(matches); matches.filter(m => m.Status === 'bye' && m.Winner).forEach(m => propagateResult(m, matches)); writeAllMatches(matches); writeConfig('Status', 'in_progress'); writeConfig('GeneratedAt', new Date().toISOString()); refreshResultFormChoices(); return { size, numPlayers, totalMatches: matches.length }; } function propagateResult(match, allMatches) { const byId = {}; allMatches.forEach(m => { byId[m.MatchID] = m; }); if (match.Winner && match.WinnerTo) { const target = byId[match.WinnerTo]; if (target) { if (!target.Player1) { target.Player1 = match.Winner; target.Player1Source = 'winner:' + match.MatchID; } else if (!target.Player2) { target.Player2 = match.Winner; target.Player2Source = 'winner:' + match.MatchID; } maybeActivate(target); } } if (match.Loser && match.LoserTo && match.Bracket !== 'L') { const target = byId[match.LoserTo]; if (target) { if (!target.Player1) { target.Player1 = match.Loser; target.Player1Source = 'loser:' + match.MatchID; } else if (!target.Player2) { target.Player2 = match.Loser; target.Player2Source = 'loser:' + match.MatchID; } maybeActivate(target); } } if (match.Bracket === 'GF' && match.Round === 1 && match.Winner) { const gf2 = allMatches.find(m => m.Bracket === 'GF' && m.Round === 2); const lbSideFinalist = match.Player2; if (gf2 && match.Winner === lbSideFinalist) { gf2.Player1 = match.Player1; gf2.Player2 = match.Player2; gf2.Status = 'ready'; } else if (gf2) { gf2.Status = 'void'; writeConfig('Status', 'complete'); } } if (match.Bracket === 'GF' && match.Round === 2 && match.Winner) { writeConfig('Status', 'complete'); } } function maybeActivate(m) { if (m.Status === 'pending' && m.Player1 && m.Player2) m.Status = 'ready'; } // ---------- Result application (shared by the Form trigger) ---------- function applyResult(matchId, winnerGamertag, score, requesterEmail) { if (!matchId || !winnerGamertag) throw new Error('Missing matchId or winnerGamertag.'); const matches = readMatches(); const match = matches.find(m => m.MatchID === matchId); if (!match) throw new Error('Match not found: ' + matchId); if (match.Status !== 'ready') throw new Error('Match ' + matchId + ' is not ready to report (status: ' + match.Status + ').'); const admin = isAdmin(requesterEmail); if (!admin) { const requesterGamertag = gamertagForEmail(requesterEmail); if (!requesterGamertag || (requesterGamertag !== match.Player1 && requesterGamertag !== match.Player2)) { throw new Error('Rejected: ' + requesterEmail + ' is not a participant in match ' + matchId + '.'); } } if (winnerGamertag !== match.Player1 && winnerGamertag !== match.Player2) { throw new Error('Winner "' + winnerGamertag + '" is not one of the two players in match ' + matchId + '.'); } match.Winner = winnerGamertag; match.Loser = winnerGamertag === match.Player1 ? match.Player2 : match.Player1; match.Score = score || ''; match.Status = 'reported'; match.ReportedBy = requesterEmail; match.ReportedAt = new Date().toISOString(); propagateResult(match, matches); matches.forEach(m => updateMatchRow(m)); return { matchId, winner: winnerGamertag }; } // ---------- Result reporting Form ---------- /** * Rebuilds the "Which result are you reporting?" dropdown to list every * currently-reportable outcome. Call this after generating a bracket or * applying a result — anywhere match statuses change. */ function refreshResultFormChoices() { const cfg = readConfig(); const formUrl = cfg.ResultFormUrl; if (!formUrl) return; // Form not set up yet — nothing to refresh const form = FormApp.openByUrl(formUrl); const item = form.getItems(FormApp.ItemType.LIST) .find(i => i.getTitle().trim() === RESULT_QUESTION_TITLE); if (!item) throw new Error('Could not find a Dropdown question titled "' + RESULT_QUESTION_TITLE + '" on the Form.'); const listItem = item.asListItem(); const ready = readMatches().filter(m => m.Status === 'ready'); if (!ready.length) { listItem.setChoiceValues(['No matches currently ready']); return; } const choices = []; ready.forEach(m => { choices.push(listItem.createChoice(m.Player1 + RESULT_SEPARATOR + m.Player2)); choices.push(listItem.createChoice(m.Player2 + RESULT_SEPARATOR + m.Player1)); }); listItem.setChoices(choices); } /** * Installable trigger target: Apps Script editor → Triggers → Add Trigger * Function: handleResultFormSubmission * Event source: From spreadsheet * Event type: On form submit */ function handleResultFormSubmission(e) { try { const nv = e.namedValues; // { 'Question title': ['answer'], ... } const emailKey = Object.keys(nv).find(k => k.toLowerCase().indexOf('email') !== -1); const requesterEmail = emailKey ? nv[emailKey][0] : ''; const answerKey = Object.keys(nv).find(k => k.trim() === RESULT_QUESTION_TITLE); const answer = answerKey ? nv[answerKey][0] : ''; const scoreKey = Object.keys(nv).find(k => k.trim() === SCORE_QUESTION_TITLE); const score = scoreKey ? nv[scoreKey][0] : ''; if (!answer || answer.indexOf(RESULT_SEPARATOR) === -1) { throw new Error('Could not parse the selected answer: "' + answer + '"'); } const [winner, loser] = answer.split(RESULT_SEPARATOR).map(s => s.trim()); const matches = readMatches(); const match = matches.find(m => m.Status === 'ready' && ((m.Player1 === winner && m.Player2 === loser) || (m.Player1 === loser && m.Player2 === winner))); if (!match) { throw new Error('No ready match found for "' + winner + '" vs "' + loser + '" — it may already be reported, or the bracket changed since the Form was last refreshed.'); } applyResult(match.MatchID, winner, score, requesterEmail); refreshResultFormChoices(); } catch (err) { // Visible in Apps Script → Executions, so admin can spot rejected/bad submissions. Logger.log('Form submission processing failed: ' + err); } } // ---------- Web app entry points (READ-ONLY, public) ---------- function doGet(e) { const action = (e.parameter.action || 'state'); let result; try { if (action === 'state') { const cfg = readConfig(); const matches = readMatches().map(m => { const copy = Object.assign({}, m); delete copy._row; return copy; }); result = { ok: true, config: { TournamentName: cfg.TournamentName || '', Status: cfg.Status || '', ResultFormUrl: cfg.ResultFormUrl || '' }, matches }; } else { result = { ok: false, error: 'Unknown action' }; } } catch (err) { result = { ok: false, error: String(err) }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } /** Writes are disabled on the public web app — reporting happens via the Form. */ function doPost(e) { const result = { ok: false, error: 'Direct submission is disabled. Please report results through the official Google Form.' }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } /** Run manually from the Apps Script editor to build the bracket. */ function adminGenerateBracketManually() { const out = generateBracket(); Logger.log(out); } /** Run manually if you ever need to force-refresh the Form's choices. */ function adminRefreshFormManually() { refreshResultFormChoices(); Logger.log('Form choices refreshed.'); } /** TEMPORARY DIAGNOSTIC — run this, share the log output, then delete it. */ function debugListFormItems() { const cfg = readConfig(); Logger.log('ResultFormUrl in Config: ' + cfg.ResultFormUrl); const form = FormApp.openByUrl(cfg.ResultFormUrl); Logger.log('Form title: ' + form.getTitle()); form.getItems().forEach(i => { Logger.log('ITEM title="' + i.getTitle() + '" type=' + i.getType() + ' id=' + i.getId()); }); }
